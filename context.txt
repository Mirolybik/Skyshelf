==> deploy.sh <==
#!/usr/bin/env bash
# SkyShelf — развёртывание тестового стенда одним скриптом (docker compose).
# Использование:
#   ./deploy.sh              — собрать и запустить стенд с демо-данными
#   ./deploy.sh --no-seed    — без заполнения демо-данными
#   ./deploy.sh --minio      — дополнительно поднять MinIO (APP_STORAGE=minio)
#   ./deploy.sh down         — остановить стенд (данные в volumes останутся)
#   ./deploy.sh logs         — смотреть логи
#   ./deploy.sh status       — состояние контейнеров
set -euo pipefail
cd "$(dirname "$0")"

COMPOSE="docker compose"
if ! command -v docker >/dev/null 2>&1; then
  echo "Ошибка: Docker не установлен. См. https://docs.docker.com/engine/install/" >&2
  exit 1
fi
if ! docker compose version >/dev/null 2>&1; then
  echo "Ошибка: не найден плагин docker compose." >&2
  exit 1
fi

case "${1:-up}" in
  down)      exec $COMPOSE --profile seed --profile minio down ;;
  logs)      exec $COMPOSE logs -f ;;
  status|ps) exec $COMPOSE ps ;;
esac

PROFILES=()
SEED=true
for arg in "$@"; do
  case "$arg" in
    --no-seed) SEED=false ;;
    --minio)   PROFILES+=(--profile minio); export APP_STORAGE=minio ;;
  esac
done

echo ">>> Сборка образов…"
# ВАЖНО: контейнер backend работает НЕ из COPY-слоёв образа, а через
# bind-mount ./backend/src и ./backend/node_modules (см. docker-compose.yml).
# Поэтому пересборка образа ничего не обновляет в раненом контейнере —
# старый процесс с restart: unless-stopped продолжает крутить прежний код.
# Перед сборкой backend принудительно останавливается compose-проект:
# это гарантирует, что после сборки контейнер стартует заново уже с
# актуальными исходниками с хоста.
if $COMPOSE ps -q backend 2>/dev/null | grep -q .; then
  echo ">>> Перезапуск стенда (backend будет поднят заново с актуальным кодом)…"
  $COMPOSE down --remove-orphans >/dev/null 2>&1 || true
fi
# Кэш BuildKit в этой среде неоднократно «залипал» на устаревшем снапшоте
# контекста backend, поэтому builder-prune остаётся как страховка.
docker builder prune -f >/dev/null 2>&1 || true
$COMPOSE "${PROFILES[@]}" build --no-cache backend || echo "Предупреждение: сборка backend не удалась — стенд использует исходники с хоста (bind-mount), это не критично." >&2
$COMPOSE "${PROFILES[@]}" build frontend || BUILD_FAILED=true

# Быстрая sanity-проверка: весь нужный код реально лежит на хосте
# (именно он монтируется в контейнер при запуске).
for f in src/server.js src/db/migrate-cli.js src/services/auth-service.js; do
  if [ ! -f "backend/$f" ]; then
    echo "Ошибка: отсутствует backend/$f — репозиторий неполный (git pull / git status)." >&2
    exit 1
  fi
done
if grep -Eq 'auth\.register\(' backend/src/server.js 2>/dev/null; then
  echo "Ошибка: в backend/src/server.js всё ещё вызывается auth.register() при старте." >&2
  echo "Рабочее дерево неактуально: выполните git pull (или git checkout -- backend/src/server.js)." >&2
  exit 1
fi

# Ключевая проверка: именно ./backend/src монтируется в контейнер, поэтому
# убеждаемся, что ВНУТРИ работающего контейнера лежит актуальный код.
# Признак старого кода: вызов auth.register() при старте (в актуальном
# server.js есть только new AuthController(services).register(app) —
# регистрация роутов, а не демо-пользователя).
OLD_CODE_PATTERN='auth\.register('

verify_mounted_code() {
  local tmp out
  # Если backend работает — читаем напрямую (быстрее всего).
  local cid
  cid=$($COMPOSE ps -q backend 2>/dev/null | head -n1)
  if [ -n "$cid" ] && docker exec "$cid" test -f /app/src/server.js 2>/dev/null; then
    out=$(docker exec "$cid" grep -Ec "$OLD_CODE_PATTERN" /app/src/server.js 2>/dev/null) || out="ERR"
    [ "$out" = "ERR" ] && return 2
    [ "$out" != "0" ]
    return
  fi
  # Иначе (контейнер в цикле рестартов) запускаем ВРЕМЕННЫЙ контейнер на
  # том же бинде, читаем из него файл и сразу удаляем.
  tmp=$($COMPOSE run -d --name skyshelf-mntcheck --no-deps --entrypoint sh backend -c 'sleep 60' 2>/dev/null) || {
    echo "Ошибка: не удалось запустить временный контейнер для проверки монтирования." >&2
    return 2
  }
  out=$(docker exec "$tmp" grep -Ec "$OLD_CODE_PATTERN" /app/src/server.js 2>/dev/null) || out="ERR"
  docker rm -f "$tmp" >/dev/null 2>&1 || true
  if [ "$out" = "ERR" ]; then
    echo "Ошибка: не удалось прочитать /app/src/server.js внутри контейнера." >&2
    return 2
  fi
  [ "$out" != "0" ]
}

echo ">>> Миграции базы данных…"
# Ждём готовности PostgreSQL (на случай свежего тома)
$COMPOSE up -d db
for _ in $(seq 1 30); do
  if $COMPOSE exec -T db pg_isready -U skyshelf -d skyshelf >/dev/null 2>&1; then break; fi
  sleep 2
done

# Миграции выполняются НЕ из собранного compose-образа, а через bind-mount:
# актуальные ./backend/src и ./backend/node_modules монтируются напрямую в
# контейнер. Так результат не зависит от кэша BuildKit (в вашей среде кэш-слой
# COPY «залип» на старом снапшоте без migrate-cli.js — см. предыдущие запуски).
DB_URL_ENV="${DATABASE_URL:-postgres://${DB_USER:-skyshelf}:${DB_PASSWORD:-skyshelf}@db:5432/${DB_NAME:-skyshelf}}"

if [ ! -d backend/node_modules/pg ]; then
  echo ">>> Установка зависимостей backend (локально, для миграций)…"
  (cd backend && npm ci --omit=dev)
fi

echo ">>> Применение миграций…"
docker run --rm \
  --network skyshelf_default \
  -v "$PWD/backend/src:/app/src:ro" \
  -v "$PWD/backend/node_modules:/app/node_modules:ro" \
  -e DATABASE_URL="$DB_URL_ENV" \
  -w /app \
  --entrypoint node \
  node:20-alpine src/db/migrate-cli.js up

echo ">>> Поднимаем стенд…"
# Если контейнер уже работает, но внутри него старый код (проверка по
# mount-указателю /app/src/server.js) — пересоздаём контейнеры заново.
if verify_mounted_code; then
  echo ">>> Внутри монтирования лежит СТАРЫЙ server.js (виден auth.register) — пересоздаём контейнеры…"
  docker rm -f skyshelf-mntcheck >/dev/null 2>&1 || true
  $COMPOSE down --remove-orphans >/dev/null 2>&1 || true
fi
$COMPOSE "${PROFILES[@]}" up -d --force-recreate backend frontend || {
  echo "Ошибка: контейнеры не поднялись. Последние логи backend:" >&2
  $COMPOSE logs --tail=50 backend >&2 || true
  exit 1
}

# Финальная проверка: код внутри контейнера = код с хоста
if verify_mounted_code; then
  echo "Ошибка: даже после пересоздания контейнера в /app/src лежит старый server.js." >&2
  echo "Это означает, что бинд ./backend/src указывает не на ваше рабочее дерево" >&2
  echo "(или скрипт запускается из другой копии репозитория). Диагностика:" >&2
  echo "  pwd   # должны быть в корне этого репозитория" >&2
  echo "  ls -i backend/src/server.js && grep -n 'auth.register(' backend/src/server.js   # на хосте совпадений быть не должно" >&2
  echo "  docker inspect \$(docker ps -aq --filter name=backend | head -n1) --format '{{json .Mounts}}'" >&2
  echo "  # в выводе Mounts Source должен путь == pwd/backend/src" >&2
  exit 1
fi

PORT="${STAND_PORT:-8088}"
# Проверка готовности выполняется быстро и с обратной связью по шагам,
# чтобы скрипт не «висел» без объяснений на одном сообщении.
check_health() {
  # сначала прямой запрос к nginx (стенд), затем — напрямую к backend-контейнеру
  curl -fsS --max-time 3 "http://localhost:${PORT}/api/health" >/dev/null 2>&1 && return 0
  local cid
  cid=$($COMPOSE ps -q backend 2>/dev/null | head -n1)
  [ -n "$cid" ] || return 1
  docker exec "$cid" node -e \
    "fetch('http://127.0.0.1:8080/api/health').then(r=>process.exit(r.ok?0:1)).catch(()=>process.exit(1))" \
    >/dev/null 2>&1
}
if check_health; then
  echo ">>> Backend уже отвечает — пропускаем ожидание."
else
  echo ">>> Ожидаем готовности backend…"
  ready=false
  for i in $(seq 1 15); do
    if check_health; then ready=true; break; fi
    echo "…попытка ${i}/15"
    sleep 2
  done
  if [ "$ready" != true ]; then
    # Диагностика: почему именно не отвечает — процесс, порт или прокси
    echo "Диагностика состояния контейнеров:" >&2
    $COMPOSE ps >&2 || true
    backend_cid=$($COMPOSE ps -q backend 2>/dev/null | head -n1)
    if [ -z "$backend_cid" ]; then
      echo "Контейнер backend не найден (не создан или упал)." >&2
    elif ! docker inspect -f '{{.State.Running}}' "$backend_cid" 2>/dev/null | grep -q true; then
      echo "Процесс backend НЕ запущен — смотрите логи ниже." >&2
    else
      echo "Процесс backend запущен, но /api/health недоступен." >&2
      echo "   внутри контейнера: $(docker exec "$backend_cid" node -e "fetch('http://127.0.0.1:8080/api/health').then(r=>console.log('OK',r.status)).catch(e=>console.log('FAIL',e.message))" 2>&1)" >&2
      echo "   снаружи (nginx):   $(curl -s -o /dev/null -w '%{http_code}' --max-time 3 "http://localhost:${PORT}/api/health" 2>&1 || echo 'нет соединения')" >&2
    fi
    echo "Последние логи backend:" >&2
    $COMPOSE logs --tail=50 backend >&2 || true
    exit 1
  fi
fi

if [ "$SEED" = true ]; then
  echo ">>> Демо-данные…"
  $COMPOSE --profile seed "${PROFILES[@]}" run --rm seed || true
else
  echo ">>> Без сидинга (--no-seed): аккаунтов в базе нет — зарегистрируйте их через форму или запустите ./deploy.sh без --no-seed."
fi

cat <<EOF

✔ Тестовый стенд SkyShelf запущен:
  Frontend:  http://localhost:${PORT}
  API:       http://localhost:${PORT}/api/health (nginx проксирует /api на backend)
  Демо-вход (после ./deploy.sh с сидингом):
    demo@skyshelf.local / SkyShelfDemo2026!   (OWNER)
    anna@skyshelf.local / SkyShelfDemo2026!   (MEMBER)
    ilya@skyshelf.local / SkyShelfDemo2026!   (READER)

Остановить: ./deploy.sh down     Логи: ./deploy.sh logs
EOF

==> docs/API.md <==
# REST API SkyShelf Final 1.0

Локальный frontend: `http://localhost:5500`. Все клиентские запросы идут на тот же origin с префиксом `/api`; Java frontend-сервер проксирует их на backend `127.0.0.1:8080`.

## Сессия и CSRF

1. `GET /api/auth/csrf` создаёт сессию и возвращает `token` и имя заголовка.
2. Каждый `POST`, `PATCH` и `DELETE` должен передать токен в этом заголовке и cookie сессии.
3. После успешного входа frontend получает свежий CSRF-токен.
4. Cookie — HttpOnly и SameSite=Lax; в cloud-профиле также Secure.

```javascript
const csrf = await fetch("/api/auth/csrf", { credentials: "same-origin" }).then(r => r.json());
await fetch("/api/auth/login", {
  method: "POST",
  credentials: "same-origin",
  headers: { "Content-Type": "application/json", [csrf.header]: csrf.token },
  body: JSON.stringify({
    email: "demo@skyshelf.local",
    password: "SkyShelfDemo2026!",
    code: "123456" // только если включена 2FA
  })
});
```

Ошибки возвращаются JSON-объектом `{ "message": "…" }`. Недоступный или чужой объект обычно даёт 404, чтобы не подтверждать его существование.

## Аутентификация и профиль

| Метод | Маршрут | Назначение |
| --- | --- | --- |
| GET | `/api/health` | `status`, версия приложения и алгоритм шифрования |
| GET | `/api/auth/csrf` | Получить CSRF-токен |
| POST | `/api/auth/register` | `name`, `email`, `password` 10–128 символов |
| POST | `/api/auth/login` | `email`, `password`, необязательный `code` TOTP |
| POST | `/api/auth/logout` | Завершить серверную сессию |
| GET | `/api/me` | Профиль без хешей и секретов |
| PATCH | `/api/me` | `name`, `theme` dark/light, `accent` blue/violet/mint |

## Личное и проектное пространство

`GET /files`, `/folders`, `/storage` без параметров работает с личным пространством. Параметр `?projectId={uuid}` переключает запрос на проект и требует членства.

### Файлы

| Метод | Маршрут | Тело / результат |
| --- | --- | --- |
| GET | `/api/files` | Список файлов, включая корзину |
| POST | `/api/files` | multipart: `file`, необязательные `folderId`, `projectId`; 201 |
| GET | `/api/files/{id}/download` | Расшифрованные байты текущей версии как attachment |
| PATCH | `/api/files/{id}` | Необязательные `name`, `folderId`, `favorite`, `trashed` |
| DELETE | `/api/files/{id}` | Переместить в корзину и отозвать ссылки |
| DELETE | `/api/files/{id}/permanent` | Физически удалить файл и все версии; только из корзины |
| GET | `/api/storage` | Тариф, квота, занятое место, число файлов и алгоритм шифрования |

DTO файла содержит: `id`, `projectId`, `name`, `size`, `mimeType`, `checksum`, `versionCount`, `folderId`, `favorite`, `trashed`, `createdAt`, `updatedAt`. Внутренний ключ объекта никогда не отдаётся клиенту.

Одна загрузка — до 20 МБ. Сервер проверяет расширение и базовую сигнатуру. Разрешены TXT, MD, CSV, PDF, PNG, JPG/JPEG, ZIP, DOCX, XLSX и PPTX.

### Версии

| Метод | Маршрут | Назначение |
| --- | --- | --- |
| GET | `/api/files/{id}/versions` | История сверху вниз: номер, автор, заметка, SHA‑256 и дата |
| POST | `/api/files/{id}/versions` | multipart: `file`, необязательный `note`; расширение должно совпасть |
| GET | `/api/files/{id}/versions/{versionId}/download` | Скачать конкретную версию |
| POST | `/api/files/{id}/versions/{versionId}/restore` | Создать новую текущую версию из выбранной старой |

Все версии шифруются независимо и учитываются в квоте. Восстановление не удаляет историю.

### Папки

| Метод | Маршрут | Назначение |
| --- | --- | --- |
| GET | `/api/folders` | Папки пространства |
| POST | `/api/folders` | `name`, `color`, необязательный `projectId` |
| PATCH | `/api/folders/{id}` | `name`, `color` |
| DELETE | `/api/folders/{id}` | Удалить только пустую папку |

Цвета: `blue`, `violet`, `mint`, `amber`, `cyan`, `rose`.

## Защищённые ссылки на файл

| Метод | Маршрут | Назначение |
| --- | --- | --- |
| GET | `/api/files/{id}/share` | Метаданные активной ссылки без исходного токена |
| POST | `/api/files/{id}/share` | `hours` 1–720, `password` 4–64 или пусто, `maxDownloads` 0–1000 |
| DELETE | `/api/files/{id}/share` | Отозвать ссылки файла |
| GET | `/api/public/{token}` | Публичные метаданные файла |
| GET | `/api/public/{token}/download` | Скачать файл, если у ссылки нет пароля |
| POST | `/api/public/{token}/download` | `{ "password": "…" }` для защищённой ссылки |

`POST /share` возвращает исходный `token` **только один раз**. Поле `id` — его SHA‑256-представление, сохранённое в БД. После достижения `maxDownloads`, истечения срока, отзыва или перемещения файла в корзину публичный запрос получает 404.

## Коды ответа

- `200/201` — успешно;
- `400` — нарушено бизнес-правило или формат;
- `401` — нет входа, неверный пароль или TOTP;
- `403` — роль не разрешает действие или не прошёл CSRF;
- `404` — объект не существует либо скрыт политикой доступа;
- `409` — конфликт уникальности;
- `413` — файл/запрос превышает лимит;
- `429` — превышен лимит попыток входа;
- `500` — например, не прошла криптографическая проверка целостности.

# Архитектура SkyShelf Final 1.0

## Подход

SkyShelf построен как **модульный монолит** с отдельным frontend. Для учебного проекта это даёт простой запуск и понятную отладку, но сохраняет границы предметных модулей:

- Authentication — регистрация, сессии, профиль и лимит входов;
- Storage — файлы, папки, версии, квоты, корзина и ссылки;
- Projects — роли, кастомизация, чек‑лист, итоговый файл и сдача;
- Security/Audit — криптография, 2FA, смена пароля и журнал;
- Admin — пользователи, блокировки, статистика и тестовые тарифы.

Frontend не обращается к БД или объектам напрямую. Он отправляет REST-запросы в Java backend, где на каждом запросе заново проверяются сессия, роль и принадлежность объекта.

```mermaid
flowchart TD
    U["Браузер пользователя"] --> F["HTML · CSS · JavaScript"]
    F -->|"same-origin /api + CSRF"| B["Spring Boot REST"]
    B --> S["Auth · Storage · Projects · Audit"]
    S --> D["H2 локально / PostgreSQL в cloud"]
    S --> O["Зашифрованные объекты: disk / MinIO"]
```

## Два режима исполнения

| Аспект | Локальная демонстрация | Docker/cloud-профиль |
| --- | --- | --- |
| Метаданные | H2 file database | PostgreSQL 17 |
| Файлы | `backend/data/objects`, не публикуется | приватный bucket MinIO |
| Frontend proxy | `FrontendServer.java` | Nginx |
| Адрес | `127.0.0.1:5500` | `127.0.0.1:8088` в compose |
| Ключ | учебный локальный ключ | обязательный `SKYSHELF_MASTER_KEY` |
| Назначение | показ без инфраструктуры | стенд, близкий к целевой архитектуре |

Оба режима используют одинаковые контроллеры, сервисы, RBAC и криптографический формат объектов.

## Поток файла

```mermaid
sequenceDiagram
    participant UI as Frontend
    participant API as StorageService
    participant DB as Metadata DB
    participant OBJ as Object store
    UI->>API: multipart + session + CSRF
    API->>API: validate type, role, quota
    API->>API: SHA-256(plain)
    API->>API: AES-256-GCM(key, random nonce)
    API->>OBJ: SKY2 | nonce | ciphertext | tag
    API->>DB: metadata + version + checksum
    DB-->>API: commit
    API-->>UI: safe file DTO
```

При скачивании порядок обратный: проверка доступа → чтение объекта → проверка тега AES-GCM → вычисление и сравнение SHA‑256 → выдача `Content-Disposition: attachment`. Если проверка целостности не проходит, байты клиенту не возвращаются.

## Модель данных

| Сущность | Назначение | Важные поля |
| --- | --- | --- |
| Account | Аккаунт и настройки | email, Argon2id hash, role, plan, 2FA state |
| Project | Учебное пространство | owner, course, teacher, deadline, visual style, final file |
| Membership | Связь пользователь–проект | projectId, userId, OWNER/MEMBER/READER |
| Folder | Папка одного уровня | owner/project, name, color |
| StoredFile | Текущая версия файла | safe objectKey, size, SHA‑256, favorite, trash |
| FileVersion | Неизменяемая версия | independent objectKey, version, author, note, checksum |
| ShareLink | Публичная ссылка файла | hashed token, Argon2id password, expiry, counter |
| ProjectTask | Пункт чек‑листа | title, completed, assignee, dueAt, position |
| SubmissionLink | Карточка преподавателю | hashed token, expiry, views |
| AuditEvent | Событие | actor, project, action, target, safe details, time |

```mermaid
erDiagram
    ACCOUNT ||--o{ MEMBERSHIP : joins
    PROJECT ||--o{ MEMBERSHIP : contains
    PROJECT ||--o{ PROJECT_TASK : tracks
    PROJECT ||--o{ STORED_FILE : owns
    STORED_FILE ||--|{ FILE_VERSION : keeps
    STORED_FILE ||--o{ SHARE_LINK : exposes
    PROJECT ||--o{ SUBMISSION_LINK : submits
```

В локальном режиме Hibernate использует `ddl-auto=update`. Для публичного production-стенда нужны версионированные миграции Flyway/Liquibase и явные ограничения БД.

ADMIN — системная роль управления метаданными, а не «суперпользователь файлов». Чтобы читать содержимое проекта, администратор должен быть добавлен обычным участником.

## Квота и конкурентность

Перед записью берётся pessimistic write lock строки Account или Project. Под этой же блокировкой сервер рассчитывает занятый объём и проверяет лимит. Это не даёт двум параллельным загрузкам одновременно пройти на последнее свободное место.

В квоту входят:

- активные файлы;
- файлы в корзине;
- все версии.

START — 15 ГБ, PERSONAL — 100 ГБ, TEAM — 500 ГБ. Базовый проект допускает пять участников, TEAM — 25.

Текущая реализация рассчитывает занятое место по объектам приложения и подходит для учебного объёма. Следующий шаг масштабирования — SQL-агрегации, индексы, пагинация и потоковая обработка больших файлов.

## Согласованность БД и объектов

Запись в БД и объектное хранилище не образует общую распределённую транзакцию. Реализована компенсация:

- если транзакция загрузки откатилась, новый объект удаляется;
- после успешного окончательного удаления метаданных очищаются объекты всех версий;
- ошибки фоновой очистки попадают в серверный журнал.

Для production нужен периодический reconciliation job, который ищет потерянные и осиротевшие объекты.

## Изоляция frontend

- клиент получает только DTO, без objectKey, password hash и секретов;
- пользовательские строки вставляются через HTML-escaping;
- файлы не обслуживаются статическим сервером;
- public download всегда attachment;
- CSP запрещает внешние скрипты, объекты и встраивание страницы;
- анимации отключаются при `prefers-reduced-motion`.

## Принятые компромиссы

- папки одноуровневые;
- одна активная публичная ссылка на файл и одна преподавательская ссылка на проект;
- до 20 МБ на файл, байты временно находятся в памяти backend;
- аудит хранится в основной БД и не является неизменяемым внешним SIEM-журналом;
- локальный HTTP предназначен только для loopback-демонстрации;
- нет антивирусного движка, восстановления пароля и управления всеми активными сессиями;
- файловое шифрование серверное, не E2EE.

Эти границы не скрываются: они вынесены в [SECURITY.md](SECURITY.md) и [PLAN.md](PLAN.md) как отдельные следующие работы.

# Защита и криптография SkyShelf

Документ объясняет не только меры, но и причину их выбора. Он соответствует реализации Final 1.0 и отделяет подтверждённые механизмы от требований будущего production-стенда.

## Что защищаем

| Актив | Основные риски | Реализованная защита |
| --- | --- | --- |
| Пароль пользователя | утечка БД, перебор | Argon2id + соль, лимит попыток, 2FA |
| Содержимое файла | чтение диска/bucket, подмена | AES‑256‑GCM + SHA‑256 + RBAC |
| Секрет 2FA | утечка БД | AES‑256‑GCM |
| Публичная ссылка | утечка токена, повторное использование | 256-bit token, hash в БД, expiry, пароль, лимит, отзыв |
| Сессия | фиксация, CSRF, кража через JS | новый session ID, CSRF, HttpOnly, SameSite |
| Проект команды | превышение полномочий | OWNER/MEMBER/READER на backend |
| Журнал действий | отсутствие трассируемости | audit events без содержимого файлов |

## Шифрование файлов

Алгоритм: **AES‑256‑GCM**.

Почему выбран GCM:

- одновременно обеспечивает конфиденциальность и аутентичность;
- подмена шифртекста или тега приводит к отказу расшифрования;
- поддерживается стандартным Java Cryptography API;
- широко применим и легко объясним на защите проекта.

Формат объекта:

```text
SKY2 (4 bytes) | random nonce (12 bytes) | ciphertext + GCM tag (16 bytes)
```

Для каждого шифрования генерируется новый nonce через `SecureRandom`. Строка `SKY2` используется и как magic header, и как AAD. Она позволяет отличить новый формат и мигрировать объекты старого MVP.

Ключ — ровно 32 байта, передаётся в Base64. Ключ не хранится рядом с объектом. В локальной демонстрации есть явно помеченный учебный ключ для запуска без настройки; в cloud-профиле переменная `SKYSHELF_MASTER_KEY` обязательна.

### Контроль целостности

Кроме GCM-tag вычисляется SHA‑256 открытых байтов. Контрольная сумма хранится в метаданных версии и сравнивается перед каждой выдачей постоянным по времени сравнением. SHA‑256 нужен для:

- понятной пользователю идентификации результата;
- проверки восстановленной версии и ZIP-пакета;
- демонстрации целостности в карточке преподавателя;
- дополнительного обнаружения рассогласования БД и объекта.

SHA‑256 не заменяет GCM и не используется как шифрование.

## Пароли

Пароли аккаунтов и пароли публичных ссылок обрабатываются **Argon2id** через Spring Security. В БД хранится только параметризованный хеш с солью.

Почему Argon2id:

- memory-hard функция усложняет массовый перебор на GPU/ASIC;
- вариант `id` сочетает устойчивость к атакам по памяти и побочным каналам;
- Spring Security предоставляет готовую проверенную реализацию.

Минимальная длина пароля аккаунта — 10 символов. Backend не возвращает хеши ни пользователю, ни администратору.

## TOTP-2FA

Реализован RFC 6238: HMAC-SHA1, 6 цифр, период 30 секунд, окно ±1 периода. Он совместим с Google Authenticator, Microsoft Authenticator, 1Password и другими TOTP-приложениями.

Порядок подключения:

1. сервер генерирует 160-битный Base32-секрет;
2. пользователь получает его и `otpauth://` URI один раз;
3. секрет сохраняется только как AES‑GCM ciphertext;
4. пользователь подтверждает текущим кодом;
5. следующие входы требуют пароль и TOTP.

Отключение требует текущий пароль и TOTP. Recovery codes пока не реализованы, поэтому для публичной эксплуатации нужен отдельный безопасный сценарий восстановления.

## Публичные токены

Для ссылки генерируется 32 случайных байта и кодируется Base64URL. В БД сохраняется только `SHA-256(rawToken)`. Исходный токен возвращается создателю один раз и находится только в URL.

Доступ можно ограничить:

- сроком 1–720 часов;
- паролем 4–64 символа;
- количеством скачиваний 1–1000, либо 0 без лимита;
- ручным отзывом;
- перемещением файла в корзину;
- блокировкой владельца.

Проверка счётчика выполняется под блокировкой записи ссылки, поэтому два одновременных запроса не должны превысить лимит.

## Сессии и браузер

- аутентификация хранится в серверной HttpSession;
- после входа меняется session ID;
- cookie HttpOnly и SameSite=Lax;
- в cloud-профиле cookie Secure;
- state-changing запросы требуют CSRF-токен;
- после выхода сессия инвалидируется;
- бездействующая сессия истекает через 30 минут.

Заголовки frontend/backend:

- Content-Security-Policy;
- `frame-ancestors 'none'` / X-Frame-Options DENY;
- X-Content-Type-Options `nosniff`;
- Referrer-Policy;
- Permissions-Policy;
- Cross-Origin-Resource-Policy.

## Доступ к проекту

- OWNER: все операции, участники, итоговый файл, публичные ссылки и оформление;
- MEMBER: файлы, версии, папки и чек‑лист;
- READER: список, чек‑лист и скачивание;
- ADMIN: служебные метаданные и тарифы, но не содержимое чужих файлов.

Проверка выполняется в сервисах, а не только скрытием кнопок. Для неизвестного/чужого объекта возвращается единый 404.

## Файловый ввод

- имя очищается и не может содержать `/`, `\`, управляющие символы, `.` или `..`;
- объект получает случайный UUID, клиентское имя не становится путём;
- разрешён только белый список расширений;
- PDF, PNG, JPEG и ZIP-контейнеры проходят базовую magic-byte проверку;
- пустые файлы и несоответствие расширению отклоняются;
- размер ограничен 20 МБ на frontend, proxy и backend;
- архивы не распаковываются;
- скачивание идёт как attachment с `no-store` и `nosniff`.

Это снижает риск, но не заменяет антивирус и Content Disarm & Reconstruction.

## Аудит

Записываются вход, профиль, загрузка/версия/удаление, папки, ссылки, участники, задачи, итоговый файл, 2FA, пароль и тарифы. Событие содержит actor, project, тип действия, target id, безопасное краткое описание и UTC-время.

Журнал намеренно не хранит содержимое файлов, пароли, исходные токены или TOTP-секреты.

## Управление ключом

Генерация ключа на Linux/macOS/Git Bash:

```bash
openssl rand -base64 32
```

PowerShell без OpenSSL:

```powershell
$bytes = New-Object byte[] 32
[Security.Cryptography.RandomNumberGenerator]::Fill($bytes)
[Convert]::ToBase64String($bytes)
```

Правила production:

- отдельный ключ для каждого окружения;
- секрет только в secret manager/переменной окружения;
- не добавлять `.env` в Git, отчёт или презентацию;
- резервная копия ключа хранится отдельно от объектов;
- потеря ключа означает потерю возможности расшифровать данные;
- ротация должна выполнять переупаковку объектов и секретов; автоматической ротации в Final 1.0 нет.

## Честные границы Final 1.0

Не реализованы или не подтверждены для публичного production:

- end-to-end/client-side encryption;
- HTTPS-сертификат и внешний deployment;
- автоматическая ротация master key;
- recovery codes и восстановление забытого пароля;
- управление всеми активными сессиями;
- антивирус/песочница для документов;
- неизменяемый внешний audit log/SIEM;
- секретное хранилище уровня KMS/HSM;
- независимый penetration test и dependency scanning;
- потоковое шифрование файлов больше 20 МБ;
- проверенный Docker restore для PostgreSQL + MinIO.

Локальный запуск слушает loopback и предназначен для демонстрации. Для публикации обязательны TLS, `APP_DEMO=false`, новый master key, недемонстрационные пароли, доверенная proxy-конфигурация, rate limiter по реальному адресу и испытанное восстановление.

Дата финального технического прогона: **26 сентября 2026 года**. Проверки выполнялись на JDK 17.0.20 и Node.js 24 в изолированных временных каталогах; пользовательская папка `backend/data` не изменялась.

## Подтверждённый результат

Исходники backend скомпилированы в 66 классов и упакованы в исполняемый Spring Boot JAR. Manifest содержит `Implementation-Version: 1.0.0`. Синтаксическая проверка прошла для `frontend/app.js`, `tests/http-smoke.cjs` и `tests/browser-smoke.cjs`.

Команда:

```powershell
node tests/http-smoke.cjs
```

Результат:

```text
PASS Final 1.0: proxy/CSP, CSRF sessions, encrypted objects, checksums,
versions/restore, favorites, project checklist/final/submission/ZIP,
protected one-use links, TOTP 2FA, audit, roles, trash, admin privacy,
tariff and backup/restore.
```

## Интеграционный набор JUnit

`backend/src/test/java/ru/skyshelf/AcceptanceTest.java` содержит 21 тестовый сценарий:

- guest/admin access и CSRF;
- регистрация, сессия и logout;
- файловый lifecycle;
- private/project role isolation;
- expiry/revoke/blocking ссылок;
- AES-GCM и проверка целостности;
- версии;
- защищённая ссылка и лимит;
- checklist/final/teacher submission;
- TOTP;
- admin plans;
- опасные имена, сигнатуры и расширения;
- foreign folder/workspace;
- лимит участников;
- две гонки квоты;
- учёт корзины;
- throttling плохих входов.

В текущем рабочем окружении Maven отсутствовал, поэтому обновлённый набор из 21 JUnit-теста здесь **не запускался командой `mvn test`**. Производственные исходники скомпилированы напрямую против зависимостей готового JAR, а основные новые сценарии подтверждены отдельным HTTP smoke. На компьютере команды перед сдачей обязательно выполнить:

```powershell
cd backend
mvn clean test
```

Предыдущая версия проекта до расширения успешно проходила 16 Maven-тестов; это не подменяет повторный прогон текущих 21.

## Настоящий браузер

`tests/browser-smoke.cjs` подготовлен для Playwright и проверяет desktop/mobile, реальные клики, скачивание, ссылку, корзину, роли, тему, admin и перезапуск backend. В текущем окружении Chromium не удалось установить из-за ограничения загрузки браузерного бинарника, поэтому визуальный прогон не объявляется выполненным.

На Windows команды:

```powershell
npm install --no-save playwright
npx playwright install chromium
node tests/browser-smoke.cjs
```

После прогона проверить:

- папку `docs/screenshots`;
- отсутствие `pageerror` и ошибок console;
- ширину 1440 и мобильный viewport 390 × 844;
- landing после полной прокрутки;
- модальные окна версий, ссылок, 2FA и участников;
- публичные страницы в приватном контексте.

## Ещё не подтверждено

- `start.bat` на конкретных компьютерах команды;
- Docker Compose с PostgreSQL и MinIO;
- согласованный backup/restore PostgreSQL + MinIO;
- публичный HTTPS-стенд;
- production rate limiting за reverse proxy;
- антивирусная проверка;
- нагрузка и большие объёмы;
- независимый security review/pentest;
- пользовательские тесты 3–5 студентов.

Эти пункты внесены в [PLAN.md](PLAN.md) и не выдаются за завершённые.

## Ручной security-check перед защитой

1. Убедиться, что `.env` и `backend/data` не попали в репозиторий/архив сдачи с личными данными.
2. Проверить, что cloud key отличается от учебного локального ключа.
3. Не публиковать демо-аккаунты на внешнем стенде.
4. Открыть protected link с неверным паролем и после лимита.
5. Войти READER и проверить отсутствие кнопок записи плюс прямой 403 на изменение.
6. Убедиться, что ADMIN без членства получает 404 при скачивании чужого файла.
7. Остановить сервер, сделать копию, восстановить в другой каталог и сравнить скачанные байты.

async function api(path, options = {}) {
  const headers = { ...options.headers };
  if (options.body && !(options.body instanceof FormData)) {
    headers["Content-Type"] = "application/json";
    options.body = JSON.stringify(options.body);
  }
  if (options.method && !["GET", "HEAD"].includes(options.method)) {
    if (!state.csrf) await csrf();
    headers[state.csrf.header] = state.csrf.token;
  }
  let res;
  try {
    res = await fetch("/api" + path, {
      ...options,
      headers,
      credentials: "same-origin",
    });
  } catch {
    throw new Error("Сервер недоступен. Проверьте, что backend запущен.");
  }
  const data = await res.json().catch(() => ({}));
  if (!res.ok) {
    if (res.status === 401 && !path.includes("/auth/")) state.user = null;
    throw new Error(data.message || `Ошибка запроса (${res.status})`);
  }
  return data;
}
async function csrf() {
  const res = await fetch("/api/auth/csrf", { credentials: "same-origin" });
  if (!res.ok) throw new Error("Не удалось подключиться к серверу.");
  state.csrf = await res.json();
}

function fileType(f) {
  const ext = f.name.split(".").pop().toUpperCase();
  const cl = ["PNG", "JPG", "JPEG"].includes(ext)
    ? "image"
    : ext === "PDF"
      ? "pdf"
      : ext === "ZIP"
        ? "zip"
        : "";
  return `<span class="file-type ${cl}">${esc(ext.slice(0, 4))}</span>`;
}

function renderFiles() {

  const list = state.files.filter(
    (f) =>
      f.trashed === trash &&
      (state.view !== "favorites" || f.favorite) &&
      (!state.folderId || f.folderId === state.folderId) &&
      f.name.toLowerCase().includes(state.query.toLowerCase()) &&
      (state.filter === "all" ||
        (state.filter === "images") === /\.(png|jpe?g)$/i.test(f.name)),
  );
  $("#file-count").textContent = list.length;
  if (!list.length) {
    $("#file-list").innerHTML = empty(
      state.query

        : trash

function formModal(title, fields, onSave, extra = "") {
  openModal(
    title,
    `<form class="dialog-form" id="dialog-form">${fields}<p class="form-error" role="alert"></p><div class="dialog-actions"><button type="button" class="btn ghost" data-action="modal-close">Отмена</button><button class="btn primary" type="submit">Сохранить</button></div>${extra}</form>`,
  );
  $("#dialog-form").onsubmit = async (e) => {
    e.preventDefault();
    const form = e.currentTarget,
      submit = $("button[type=submit]", form);
    submit.disabled = true;
    try {
      await onSave(Object.fromEntries(new FormData(form)));
      modal.close();
      await refresh();
    } catch (e) {
      $(".form-error", form).textContent = e.message;
    } finally {
      submit.disabled = false;
    }
  };
}

async function uploadFiles(files) {
  if (state.uploading) return;
  if (!writable()) return toast("У вас доступ только для чтения.", true);
  const selected = [...files];
  if (!selected.length) return;
  state.uploading = true;
  let done = 0;
  const projectId = state.projectId,
    folderId = state.folderId;
  try {
    for (const file of selected) {
      if (file.size > 20 * 1024 * 1024)
        throw new Error(`${file.name}: размер больше 20 МБ.`);
      const status = $("#upload-status");
      if (status)
        status.textContent = `Загружаем ${done + 1} из ${selected.length}: ${file.name}`;
      const form = new FormData();
      form.append("file", file);
      if (folderId) form.append("folderId", folderId);
      if (projectId) form.append("projectId", projectId);
      await api("/files", { method: "POST", body: form });
      done++;
    }
    toast(`Загружено файлов: ${done}`);
  } catch (e) {
    toast(`${e.message}${done ? " Уже загружено: " + done : ""}`, true);
  } finally {
    state.uploading = false;
    $("#upload-input").value = "";
    await refresh();
  }
}
function setupDrop() {
  const zone = $(".drop-zone");
  if (!zone) return;
  zone.onkeydown = (e) => {
    if (e.key === "Enter" || e.key === " ") {
      e.preventDefault();
      $("#upload-input").click();
    }
  };
  zone.ondragover = (e) => {
    e.preventDefault();
    zone.classList.add("dragover");
  };
  zone.ondragleave = () => zone.classList.remove("dragover");
  zone.ondrop = (e) => {
    e.preventDefault();
    zone.classList.remove("dragover");
    uploadFiles(e.dataTransfer.files);
  };
}

document.addEventListener("click", async (event) => {
  const target = event.target.closest("[data-action]");
  if (!target) return;
  const { action, id } = target.dataset;
  try {
    switch (action) {
      case "scroll-section": {
        const section=document.getElementById(target.dataset.section);if(section){event.preventDefault();section.scrollIntoView({behavior:"smooth",block:"start"});}else location.hash="#home";break;
      }
      case "menu":
        $(".sidebar").classList.toggle("open");
        break;
      case "modal-close":
        modal.close();
        break;
      case "logout":
        await api("/auth/logout", { method: "POST" });
        state.user = null;
        state.csrf = null;
        applyAppearance();
        location.hash = "#home";
        toast("Вы вышли из аккаунта");
        break;
      case "upload":
        $("#upload-input").click();
        break;
      case "filter":
        state.filter = target.dataset.filter;
        if(state.view==="project") await projectPage(); else workspacePage();
        break;
      case "open-folder":
        state.folderId = id;
        if(state.view==="project") await projectPage(); else workspacePage();
        break;
      case "folder-back":
        state.folderId = null;
        if(state.view==="project") await projectPage(); else workspacePage();
        break;
      case "folder-new":
        folderDialog();
        break;
      case "folder-edit":
        folderDialog(id);
        break;
      case "folder-delete":
        confirmModal(
          "Удалить папку?",
          "Удалить можно только пустую папку. Файлы из неё, включая корзину, сначала нужно переместить.",
          async () => {
            await api(`/folders/${id}`, { method: "DELETE" });
            toast("Папка удалена");
          },
        );
        break;
      case "file-edit":
        fileDialog(id);
        break;
      case "favorite": {
        const file=state.files.find(f=>f.id===id);await api(`/files/${id}`,{method:"PATCH",body:{favorite:!file.favorite}});await refresh();toast(file.favorite?"Убрано из избранного":"Добавлено в избранное");break;
      }
      case "versions":
        await versionsDialog(id);

/* Прикладная криптография: AES-256-GCM для объектов и секретов, SHA-256 для целостности
   и токенов ссылок, Argon2id для паролей. Порт CryptoService.java. */
'use strict';

const crypto = require('node:crypto');
const argon2 = require('argon2');

const MAGIC = Buffer.from('SKY2'); // конверт нового формата: MAGIC | nonce(12) | ciphertext+tag
const NONCE_SIZE = 12;

class CryptoService {
  constructor(masterKeyBase64) {
    const raw = Buffer.from(masterKeyBase64, 'base64');
    if (raw.length !== 32) throw new Error('SKYSHELF_MASTER_KEY must contain a Base64 encoded 32-byte key');
    this.key = raw;
  }

  encrypt(plain) {
    const nonce = crypto.randomBytes(NONCE_SIZE);
    const cipher = crypto.createCipheriv('aes-256-gcm', this.key, nonce);
    cipher.setAAD(MAGIC);
    const encrypted = Buffer.concat([cipher.update(plain), cipher.final()]);
    return Buffer.concat([MAGIC, nonce, encrypted, cipher.getAuthTag()]);
  }

  /** Расшифровка конверта; объекты старого формата (без MAGIC) возвращаются как есть — порт миграции. */
  decrypt(envelope) {
    if (!this.isEncrypted(envelope)) return envelope;
    const nonce = envelope.subarray(MAGIC.length, MAGIC.length + NONCE_SIZE);
    const payload = envelope.subarray(MAGIC.length + NONCE_SIZE);
    const authTag = payload.subarray(payload.length - 16);
    const decipher = crypto.createDecipheriv('aes-256-gcm', this.key, nonce);
    decipher.setAAD(MAGIC);
    decipher.setAuthTag(authTag);
    return Buffer.concat([decipher.update(payload.subarray(0, payload.length - 16)), decipher.final()]);
  }

  isEncrypted(bytes) {
    return bytes.length > MAGIC.length + NONCE_SIZE + 16 && bytes.subarray(0, MAGIC.length).equals(MAGIC);
  }

  encryptText(value) { return this.encrypt(Buffer.from(value, 'utf8')).toString('base64'); }
  decryptText(value) { return this.decrypt(Buffer.from(value, 'base64')).toString('utf8'); }

  static checksum(buffer) { return crypto.createHash('sha256').update(buffer).digest('hex'); }

  /** Токены ссылок: в БД хранится только URL-safe base64 от SHA-256. */
  static tokenHash(token) {
    return crypto.createHash('sha256').update(String(token ?? ''), 'utf8').digest('base64url');
  }

  newToken() { return crypto.randomBytes(32).toString('base64url'); }

  hashPassword(password) { return argon2.hash(password, { type: argon2.argon2id }); }
  verifyPassword(hash, password) {
    return argon2.verify(hash, password ?? '').catch(() => false);
  }

  /** Constant-time сравнение строк контрольных сумм. */
  static safeEquals(a, b) {
    const ba = Buffer.from(String(a), 'ascii'), bb = Buffer.from(String(b), 'ascii');
    return ba.length === bb.length && crypto.timingSafeEqual(ba, bb);
  }
}

module.exports = { CryptoService, MAGIC };

    this.accounts = accounts;
  }

    const clean = details == null ? '' : String(details).trim().slice(0, 500);
    await this.audits.insert(new AuditEvent({ actorId, projectId, action, targetType, targetId, details: clean }));
  }

  async visible(userId, requested) {
    const limit = Math.max(1, Math.min(100, Number(requested) || 50));
    const memberships = await this.memberships.listByUser(userId);
    const projectIds = memberships.map((m) => m.projectId);
    const events = await this.audits.listVisible(userId, projectIds, limit);
    const names = new Map();
    for (const e of events) {
      if (e.actorId && !names.has(e.actorId)) {
        const account = await this.accounts.findById(e.actorId);
        names.set(e.actorId, account ? account.name : 'Система');
      }
    }
    return events.map((e) => ({
      id: e.id, action: e.action, targetType: e.targetType, targetId: e.targetId,
      projectId: e.projectId, details: e.details, createdAt: e.createdAt,
      actor: e.actorId ? names.get(e.actorId) ?? 'Система' : 'Система',
    }));
  }
}

/* Ролевой доступ к проектам (порт ProjectService.java): OWNER/MEMBER/READER, лимиты тарифов,

'use strict';

const { Enums, validateName } = require('../domain/entities');
const { BadRequestError, ForbiddenError, NotFoundError } = require('../domain/errors');

class ProjectService {
  constructor({ projects, memberships, accounts, tasks, files, submissions, crypto, audit }) {
    this.projects = projects;
    this.memberships = memberships;
    this.accounts = accounts;
    this.tasks = tasks;
    this.files = files;
    this.submissions = submissions;
    this.crypto = crypto;
    this.audit = audit;
  }

  static clean(value, max) { return value == null ? '' : String(value).trim().slice(0, max); }
  static validRole(role) { if (!Enums.MEMBER_ROLES.has(role)) throw new BadRequestError('Выберите роль участника или читателя.'); }
  static validAccent(value) { const v = value ?? 'blue'; if (!Enums.PROJECT_ACCENTS.has(v)) throw new BadRequestError('Неизвестный акцент проекта.'); return v; }
  static validCover(value) { const v = value ?? 'orb'; if (!Enums.PROJECT_COVERS.has(v)) throw new BadRequestError('Неизвестный стиль обложки.'); return v; }
  static validStatus(value) { if (!Enums.PROJECT_STATUSES.has(value)) throw new BadRequestError('Неизвестный статус проекта.'); return value; }
  memberLimit(project) { return project.plan === 'TEAM' ? 25 : 5; }

  /** Роль пользователя в проекте; проект/blocked-владелец недоступны → 404 (как в Java-версии). */
  async role(projectId, userId) {
    const project = await this.projects.findById(projectId);
    if (!project) throw new NotFoundError();
    const owner = await this.accounts.findById(project.ownerId);
    if (!owner || owner.blocked) throw new NotFoundError();
    const membership = await this.memberships.find(projectId, userId);
    if (!membership) throw new NotFoundError();
    return membership.role;
  }

  async access(projectId, userId, write) {
    const role = await this.role(projectId, userId);
    if (write && role === 'READER') throw new ForbiddenError('Читатель может только просматривать и скачивать файлы.');
  }

  async owner(projectId, userId) {
    if ((await this.role(projectId, userId)) !== 'OWNER') throw new ForbiddenError('Действие доступно владельцу проекта.');
  }

  async create(userId, name, course = '', teacher = '', deadline = null, accent = 'blue', cover = 'orb') {
    const project = await this.projects.insert({
      ownerId: userId, name: validateName(name, 120), course: ProjectService.clean(course, 120),
      teacher: ProjectService.clean(teacher, 120), deadline, accent: ProjectService.validAccent(accent),
      cover: ProjectService.validCover(cover),
    });
    await this.memberships.insert({ projectId: project.id, userId, role: 'OWNER' });
    await this.seedTasks(project.id);
    await this.audit.record(userId, project.id, 'PROJECT_CREATED', 'PROJECT', project.id, project.name);
    return project;
  }

  async update(id, actorId, { name, description, course, teacher, deadline, accent, cover, status }) {
    await this.projects.lockById(id);
    await this.owner(id, actorId);
    const patch = {};
    if (name != null) patch.name = validateName(name, 120);
    if (description != null) patch.description = ProjectService.clean(description, 700);
    if (course != null) patch.course = ProjectService.clean(course, 120);
    if (teacher != null) patch.teacher = ProjectService.clean(teacher, 120);
    patch.deadline = deadline === undefined ? undefined : deadline;
    if (accent != null) patch.accent = ProjectService.validAccent(accent);
    if (cover != null) patch.cover = ProjectService.validCover(cover);
    if (status != null) patch.status = ProjectService.validStatus(status);
    for (const k of Object.keys(patch)) if (patch[k] === undefined) delete patch[k];

    if (!project) throw new NotFoundError();

    const project = await this.projects.findById(projectId);

      await this.projects.update(projectId, { finalFileId: null, status: 'IN_PROGRESS' });
    }
  }

  async createSubmission(projectId, actorId, hours) {
    const project = await this.projects.lockById(projectId);
    if (!project) throw new NotFoundError();
    await this.owner(projectId, actorId);
    if (!project.finalFileId) throw new BadRequestError('Сначала отметьте итоговый файл проекта.');
    if (hours < 1 || hours > 720) throw new BadRequestError('Срок ссылки — от 1 до 720 часов.');
    await this.submissions.deleteByProject(projectId);
    const raw = this.crypto.newToken();
    const link = await this.submissions.insertSubmission({
      token: this.constructor.tokenHash(raw), projectId,
      expiresAt: new Date(Date.now() + hours * 3600_000),
    });
    await this.projects.update(projectId, { status: 'SUBMITTED' });
    await this.audit.record(actorId, projectId, 'SUBMISSION_LINK_CREATED', 'PROJECT', projectId, `Срок ${hours} ч.`);
    return { token: raw, id: link.token, expiresAt: link.expiresAt ?? null };
  }

  static tokenHash(raw) {
    // делегируем статике CryptoService, чтобы не дублировать логику
    const { CryptoService } = require('./crypto-service');
    return CryptoService.tokenHash(raw);
  }

  async submissionList(projectId, actorId) {
    await this.owner(projectId, actorId);
    const links = await this.submissions.listByProject(projectId);
    return links.map((s) => ({ id: s.token, createdAt: s.createdAt, expiresAt: s.expiresAt, views: s.views }));
  }

  async revokeSubmissions(projectId, actorId) {
    await this.owner(projectId, actorId);
    await this.submissions.deleteByProject(projectId);
    await this.audit.record(actorId, projectId, 'SUBMISSION_LINK_REVOKED', 'PROJECT', projectId, '');
  }

  /** Публичная карточка преподавателя; countView=true блокирует строку и увеличивает просмотры. */
  async publicSubmission(rawToken, countView) {
    const hash = this.constructor.tokenHash(rawToken ?? '');
    const link = countView ? await this.submissions.lockByToken(hash) : await this.submissions.findByToken(hash);
    if (!link || new Date(link.expiresAt) <= new Date()) throw new NotFoundError();
    const project = await this.projects.findById(link.projectId);
    if (!project) throw new NotFoundError();
    const owner = await this.accounts.findById(project.ownerId);
    if (!owner || owner.blocked || !project.finalFileId) throw new NotFoundError();
    const file = await this.files.findById(project.finalFileId);
    if (!file || file.trashed) throw new NotFoundError();
    if (countView) await this.submissions.bumpViews(hash);
    return { project, file, link };
  }

  async seedTasks(projectId) {
    const titles = ['Собрать материалы проекта', 'Проверить роли участников', 'Отметить итоговый файл', 'Создать ссылку преподавателю'];
    for (const [position, title] of titles.entries()) {
      await this.tasks.insert({ projectId, title, position });
    }
  }
}

/* RBAC и аутентификация (порт SecurityConfig + AuthController): проверка пароля Argon2id,
   TOTP-шаг, смена session ID при входе, троттлинг попыток входа, актуальность блокировки. */
'use strict';

const { UnauthorizedError, ForbiddenError, TooManyRequestsError, ConflictError, BadRequestError } = require('../domain/errors');

class AuthService {
  constructor({ accounts, crypto, totp, audit, sessions }) {
    this.accounts = accounts;
    this.crypto = crypto;
    this.totp = totp;
    this.audit = audit;
    this.sessions = sessions;
    this.attempts = new Map(); // key → { count, until }
  }

  static normalizeEmail(email) { return String(email ?? '').trim().toLowerCase(); }

  /** Скользящее окно на IP и email (порт limit() из AuthController). */
  limit(key, max) {
    const now = Date.now();
    for (const [k, v] of this.attempts) if (v.until < now) this.attempts.delete(k);
    if (this.attempts.size > 10_000) throw new TooManyRequestsError('Повторите попытку позже.');
    const old = this.attempts.get(key) ?? { count: 0, until: now + 600_000 };
    if (old.count >= max) throw new TooManyRequestsError('Слишком много попыток. Попробуйте через 10 минут.');
    this.attempts.set(key, { count: old.count + 1, until: old.until });
  }

  async register({ name, email, password }, ip) {
    this.limit(`register:${ip}`, 20);
    if (!name || !String(name).trim()) throw new BadRequestError('Проверьте поля: имя, корректный email и пароль от 10 символов.');
    const normalized = AuthService.normalizeEmail(email);
    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(normalized) || normalized.length > 180) throw new BadRequestError('Проверьте поля: имя, корректный email и пароль от 10 символов.');
    if (typeof password !== 'string' || password.length < 10 || password.length > 128) throw new BadRequestError('Проверьте поля: имя, корректный email и пароль от 10 символов.');
    if (await this.accounts.findByEmail(normalized)) throw new ConflictError('Этот email уже зарегистрирован.');
    const account = await this.accounts.insert({
      email: normalized,
      name: String(name).trim().slice(0, 80),
      passwordHash: await this.crypto.hashPassword(password),
    });
    await this.audit.record(account.id, null, 'ACCOUNT_REGISTERED', 'ACCOUNT', account.id, normalized);
    return account;
  }

  /** Проверка пары email/пароль без создания сессии (используется контроллером логина). */
  async authenticate(email, password, ip) {
    const normalized = AuthService.normalizeEmail(email);
    this.limit(`ip:${ip}`, 60);
    this.limit(`email:${normalized}`, 10);
    const account = await this.accounts.findByEmail(normalized);
    const provided = typeof password === 'string' ? password : '';
    const ok = account ? await this.crypto.verifyPassword(account.passwordHash, provided) : false;
    if (!account || !ok || account.blocked) {
      throw new UnauthorizedError('Неверный email или пароль, либо аккаунт заблокирован.');
    }
    if (account.twoFactorEnabled) {
      // второй шаг — код TOTP проверяет контроллер login (чтобы не светить различие состояний здесь)
    }
    this.attempts.delete(`email:${normalized}`);
    return account;
  }

  async verifyTwoFactor(account, code) {
    try {
      if (!account.twoFactorSecret) throw new UnauthorizedError('Введите корректный 6-значный код 2FA.');
      if (!this.totp.valid(this.crypto.decryptText(account.twoFactorSecret), code)) {
        throw new UnauthorizedError('Введите корректный 6-значный код 2FA.');
      }
    } catch (error) {
      if (error.status) throw error;
      throw new UnauthorizedError('Не удалось проверить двухфакторную защиту.');
    }
  }

  /** Аккаунт из запроса: сессия + актуальный бан (порт current(Authentication)). */
  async current(session) {
    if (!session || !session.userId) throw new UnauthorizedError();
    const account = await this.accounts.findById(session.userId);
    if (!account) throw new UnauthorizedError('Аккаунт не найден');
    if (account.blocked) throw new ForbiddenError('Аккаунт заблокирован администратором.');
    return account;
  }

  async requireAdmin(session) {
    const account = await this.current(session);
    if (account.role !== 'ADMIN') throw new ForbiddenError('Действие доступно администратору.');
    return account;
  }

  async touchLogin(account) {
    await this.accounts.update(account.id, { lastLoginAt: new Date() });
    await this.audit.record(account.id, null, 'LOGIN_SUCCEEDED', 'ACCOUNT', account.id, 'Вход в защищённую сессию');
  }
}

/* Личное и командное хранилище (порт StorageService.java): загрузка с прикладным шифрованием,
   версии файлов, папки, квоты тарифов, публичные ссылки, корзина и полная очистка. */
'use strict';

const crypto = require('node:crypto');
const { Enums, validateName } = require('../domain/entities');
const { CryptoService } = require('./crypto-service');
const {
  BadRequestError, UnauthorizedError, NotFoundError, ConflictError,
  PayloadTooLargeError, IntegrityError,
} = require('../domain/errors');

class StorageService {
  constructor({ files, folders, shares, accounts, versions, projectsRepo, objects, crypto: enc, audit, projectService, config }) {
    this.files = files;
    this.folders = folders;
    this.shares = shares;
    this.accounts = accounts;
    this.versions = versions;
    this.projectsRepo = projectsRepo;
    this.objects = objects;
    this.crypto = enc;
    this.audit = audit;
    this.projects = projectService;
    this.config = config;
  }

  /** Токены ссылок: в БД хранится только SHA-256 (base64url) от публичного токена. */
  static tokenHash(raw) { return CryptoService.tokenHash(raw); }
  static safeEquals(a, b) {
    const ba = Buffer.from(String(a), 'ascii'), bb = Buffer.from(String(b), 'ascii');
    return ba.length === bb.length && crypto.timingSafeEqual(ba, bb);
  }

  static ext(name) { const dot = String(name).lastIndexOf('.'); return dot < 0 ? '' : name.slice(dot + 1).toLowerCase(); }
  static mime(name) { return Enums.MIME_BY_EXT[StorageService.ext(name)] ?? 'application/octet-stream'; }

  /** Магические байты против расширения (порт validateFile). */
  static validateFile(name, bytes) {
    const ext = StorageService.ext(name);
    if (!Enums.FILE_EXTENSIONS.has(ext)) throw new BadRequestError('Разрешены TXT, MD, CSV, PDF, PNG, JPG, ZIP, DOCX, XLSX, PPTX.');
    if (bytes.length === 0) throw new BadRequestError('Пустой файл нельзя загрузить.');
    if (bytes.length > 20 * 1024 * 1024) throw new PayloadTooLargeError();
    const starts = (arr) => bytes.length >= arr.length && bytes.subarray(0, arr.length).equals(Buffer.from(arr));
    const valid = (() => {
      switch (ext) {
        case 'pdf': return starts([0x25, 0x50, 0x44, 0x46, 0x2d]); // %PDF-
        case 'png': return starts([137, 80, 78, 71, 13, 10, 26, 10]);
        case 'jpg': case 'jpeg': return bytes.length >= 3 && bytes[0] === 0xff && bytes[1] === 0xd8 && bytes[2] === 0xff;
        case 'zip': case 'docx': case 'xlsx': case 'pptx': return bytes.length >= 4 && bytes[0] === 0x50 && bytes[1] === 0x4b && [3, 5, 7].includes(bytes[2]);
        default: return !bytes.includes(0); // текст: без нулевых байтов
      }
    })();
    if (!valid) throw new BadRequestError('Содержимое файла не соответствует расширению.');
  }

  async accountName(id) {
    if (!id) return 'Участник';
    const account = await this.accounts.findById(id);
    return account ? account.name : 'Участник';
  }

  async access(actualOwner, projectId, actor, write) {
    if (projectId != null) await this.projects.access(projectId, actor, write);
    else if (actualOwner !== actor) throw new NotFoundError();
  }

  async owned(id, actor) {
    const file = await this.files.findById(id);
    if (!file) throw new NotFoundError();
    await this.access(file.ownerId, file.projectId, actor, false);
    return file;
  }

  async folder(id, actor) {
    const folder = await this.folders.findById(id);
    if (!folder) throw new NotFoundError();
    await this.access(folder.ownerId, folder.projectId, actor, false);
    return folder;
  }

  async workspaceFiles(user, projectId) {
    if (projectId != null) {
      await this.projects.access(projectId, user, false);
      return this.files.listByProject(projectId);
    }
    return this.files.listByOwner(user);
  }

  async folderList(user, projectId) {
    if (projectId != null) await this.projects.access(projectId, user, false);
    return this.folders.listFor(user, projectId);
  }

  async used(user, projectId) {
    const list = await this.workspaceFiles(user, projectId);
    let total = 0n;
    for (const f of list) {
      const versions = await this.versions.listByFile(f.id);
      total += versions.length
        ? versions.reduce((s, v) => s + BigInt(v.size), 0n)
        : BigInt(f.size);
    }
    return total;
  }

  async plan(user, projectId) {
    if (projectId == null) {
      const account = await this.accounts.findById(user);
      if (!account) throw new NotFoundError();
      return account.plan;
    }
    const project = await this.projectsRepo.findById(projectId);
    if (!project) throw new NotFoundError();
    return project.plan;
  }

  async limit(user, projectId) {
    const plan = await this.plan(user, projectId);
    return Enums.QUOTA_BYTES[plan] ?? this.config.quotaBytes;
  }

  /** Блокировка владельца/проекта на время транзакции (порт lock()). */
  async lock(user, projectId) {
    if (projectId == null) {
      const account = await this.accounts.lockById(user);
      if (!account) throw new NotFoundError();
    } else {
      const project = await this.projectsRepo.lockById(projectId);
      if (!project) throw new NotFoundError();
      await this.projects.access(projectId, user, true);
    }
  }

  async putEncrypted(key, plain) { await this.objects.put(key, this.crypto.encrypt(plain)); }

  async readObject(key, checksum) {
    const plain = this.crypto.decrypt(await this.objects.get(key));
    if (checksum && !StorageService.safeEquals(checksum, CryptoService.checksum(plain))) throw new IntegrityError();
    return plain;
  }

  async content(file) { return this.readObject(file.objectKey, file.checksum); }

  async saveVersion(file, actor, note, key, size, sum, mime, number) {
    const clean = note == null ? '' : String(note).trim().slice(0, 240);
    return this.versions.insert({
      fileId: file.id, createdById: actor, note: clean, objectKey: key,
      size: String(size), checksum: sum, mimeType: mime, versionNumber: number,
    });
  }

  /** Историческим файлам без записи версии создаём v1 (порт ensureVersionRow). */
  async ensureVersionRow(file) {
    const existing = await this.versions.listByFile(file.id);
    if (existing.length) return;
    await this.saveVersion(file, file.createdById ?? file.ownerId, 'Импортированная первая версия',
      file.objectKey, file.size, file.checksum, file.mimeType ?? StorageService.mime(file.name), 1);
    await this.files.update(file.id, { versionCount: 1 });
    file.versionCount = 1;
  }

  async upload(owner, filename, folderId, bytes, projectId = null) {
    await this.lock(owner, projectId);
    const clean = validateName(filename, 180);
    StorageService.validateFile(clean, bytes);
    if (folderId) {
      const folder = await this.folder(folderId, owner);
      if (folder.projectId !== projectId) throw new BadRequestError('Папка находится в другом пространстве.');
    } else folderId = null;
    const used = await this.used(owner, projectId);
    if (used + BigInt(bytes.length) > await this.limit(owner, projectId)) throw new BadRequestError('Недостаточно места в хранилище.');
    const objectKey = crypto.randomUUID();
    await this.putEncrypted(objectKey, bytes); // при откате транзакции объект останется осиротевшим — как in-memory cleanup у Spring
    const file = await this.files.insert({
      ownerId: owner, createdById: owner, projectId, name: clean, folderId,
      size: String(bytes.length), mimeType: StorageService.mime(clean),
      checksum: CryptoService.checksum(bytes), objectKey, versionCount: 1,
    });
    await this.saveVersion(file, owner, 'Первая версия', objectKey, bytes.length, file.checksum, file.mimeType, 1);
    await this.audit.record(owner, projectId, 'FILE_UPLOADED', 'FILE', file.id, `${clean} · SHA-256 ${file.checksum.slice(0, 12)}`);
    return file;
  }

  async addVersion(id, actor, incomingName, bytes, note) {
    const file = await this.owned(id, actor);
    await this.lock(actor, file.projectId);
    await this.access(file.ownerId, file.projectId, actor, true);
    if (file.trashed) throw new BadRequestError('Сначала восстановите файл из корзины.');
    const clean = validateName(incomingName, 180);
    if (StorageService.ext(clean) !== StorageService.ext(file.name)) throw new BadRequestError('Новая версия должна иметь то же расширение.');
    StorageService.validateFile(clean, bytes);
    const used = await this.used(actor, file.projectId);
    if (used + BigInt(bytes.length) > await this.limit(actor, file.projectId)) throw new BadRequestError('Недостаточно места: версии файлов учитываются в квоте.');
    await this.ensureVersionRow(file);
    const number = Math.max(1, file.versionCount) + 1;
    const key = crypto.randomUUID();
    const sum = CryptoService.checksum(bytes);
    await this.putEncrypted(key, bytes);
    await this.saveVersion(file, actor, note, key, bytes.length, sum, StorageService.mime(clean), number);
    await this.files.update(file.id, { objectKey: key, size: String(bytes.length), checksum: sum, mimeType: StorageService.mime(clean), versionCount: number, updatedAt: new Date() });
    await this.audit.record(actor, file.projectId, 'FILE_VERSION_ADDED', 'FILE', file.id, `${file.name} · версия ${number}`);
    return this.files.findById(file.id);
  }

  async versionList(id, actor) {
    const file = await this.owned(id, actor);
    await this.ensureVersionRow(file);
    return this.versions.listByFile(id);
  }

  async version(fileId, versionId, actor) {

  }

  async restoreVersion(fileId, versionId, actor) {

  }

  async update(id, owner, newName, targetFolder, favorite, trashed) {
    const file = await this.owned(id, owner);
    await this.lock(owner, file.projectId);
    await this.access(file.ownerId, file.projectId, owner, true);
    if (file.trashed && trashed !== false) throw new BadRequestError('Сначала восстановите файл из корзины.');
    const patch = {};
    if (newName != null) {
      const clean = validateName(newName, 180);
      if (StorageService.ext(clean) !== StorageService.ext(file.name)) throw new BadRequestError('При переименовании сохраните расширение файла.');
      patch.name = clean;
      patch.mimeType = StorageService.mime(clean);
    }
    if (targetFolder != null) {
      if (targetFolder === '') patch.folderId = null;
      else {
        const folder = await this.folder(targetFolder, owner);
        if (folder.projectId !== file.projectId) throw new BadRequestError('Папка находится в другом пространстве.');
        patch.folderId = targetFolder;
      }
    }
    if (favorite != null) patch.favorite = favorite;
    if (trashed != null) {
      patch.trashed = trashed;
      if (trashed) await this.shares.deleteByFile(file.id);
      await this.audit.record(owner, file.projectId, trashed ? 'FILE_TRASHED' : 'FILE_RESTORED', 'FILE', file.id, file.name);
    }
    patch.updatedAt = new Date();
    await this.files.update(file.id, patch);
    return this.files.findById(file.id);
  }

  static validColor(color) { if (!Enums.FOLDER_COLORS.has(color)) throw new BadRequestError('Выберите цвет папки.'); }

  }

  async share(id, owner, hours, password, maxDownloads) {
    const file = await this.owned(id, owner);
    if (file.trashed) throw new BadRequestError('Файл в корзине.');
    await this.lock(owner, file.projectId);
    if (file.projectId != null) await this.projects.owner(file.projectId, owner);
    if (hours < 1 || hours > 720) throw new BadRequestError('Срок ссылки — от 1 до 720 часов.');
    if (maxDownloads < 0 || maxDownloads > 1000) throw new BadRequestError('Лимит скачиваний — от 0 до 1000.');
    const pass = password == null ? '' : String(password).trim();
    if (pass && (pass.length < 4 || pass.length > 64)) throw new BadRequestError('Пароль ссылки: от 4 до 64 символов.');
    await this.shares.deleteByFile(id);
    const raw = this.crypto.newToken();
    const link = await this.shares.insertShare({
      token: this.constructor.tokenHash(raw), fileId: id,
      expiresAt: new Date(Date.now() + hours * 3600_000), maxDownloads,
      passwordHash: pass ? await this.crypto.hashPassword(pass) : null,
    });
    await this.audit.record(owner, file.projectId, 'SHARE_CREATED', 'FILE', file.id,
      `Срок ${hours} ч., лимит ${maxDownloads === 0 ? 'без ограничений' : maxDownloads}`);
    return { token: raw, id: link.token, expiresAt: link.expiresAt, passwordRequired: !!link.passwordHash, maxDownloads: link.maxDownloads, downloadCount: 0 };
  }

  async shareList(id, owner) {
    const file = await this.owned(id, owner);
    if (file.projectId != null) await this.projects.owner(file.projectId, owner);
    const links = await this.shares.listByFile(id);
    return links.map((s) => s.view());
  }

  async revoke(id, owner) {
    const file = await this.owned(id, owner);
    await this.lock(owner, file.projectId);
    if (file.projectId != null) await this.projects.owner(file.projectId, owner);
    await this.shares.deleteByFile(id);
    await this.audit.record(owner, file.projectId, 'SHARE_REVOKED', 'FILE', id, file.name);
  }

  async publicInfo(rawToken) {
    const link = await this.publicLink(rawToken, false, null);
    return { file: await this.availableSharedFile(link), link };
  }

  async publicDownload(rawToken, password) {
    const link = await this.publicLink(rawToken, true, password);
    const file = await this.availableSharedFile(link);
    await this.shares.updateByToken(link.token, { download_count: link.downloadCount + 1 });
    return { file, link };
  }

  async publicLink(raw, download, password) {
    const hash = this.constructor.tokenHash(raw ?? '');
    const link = download ? await this.shares.lockByToken(hash) : await this.shares.findByToken(hash);
    if (!link || new Date(link.expiresAt) <= new Date()) throw new NotFoundError();
    if (link.maxDownloads > 0 && link.downloadCount >= link.maxDownloads) throw new NotFoundError();
    if (download && link.passwordHash && !(await this.crypto.verifyPassword(link.passwordHash, password ?? ''))) {
      throw new UnauthorizedError('Неверный пароль ссылки.');
    }
    return link;
  }

  async availableSharedFile(link) {
    const file = await this.files.findById(link.fileId);
    if (!file || file.trashed) throw new NotFoundError();
    let ownerId = file.ownerId;
    if (file.projectId != null) {
      const project = await this.projectsRepo.findById(file.projectId);
      if (!project) throw new NotFoundError();
      ownerId = project.ownerId;
    }
    const owner = await this.accounts.findById(ownerId);
    if (!owner || owner.blocked) throw new NotFoundError();
    return file;
  }

  /** Полная очистка файла из корзины: версии, ссылки, объекты после коммита (порт purge). */
  async purge(id, actor) {
    const file = await this.owned(id, actor);
    await this.lock(actor, file.projectId);
    if (file.projectId != null) await this.projects.owner(file.projectId, actor);
    if (!file.trashed) throw new BadRequestError('Сначала переместите файл в корзину.');
    await this.shares.deleteByFile(id);
    const keys = new Set((await this.versions.listByFile(id)).map((v) => v.objectKey));
    keys.add(file.objectKey);
    await this.versions.deleteByFile(id);
    await this.files.deleteById(id);
    await this.projects.clearFinalIf(file.projectId, id);
    await this.audit.record(actor, file.projectId, 'FILE_PURGED', 'FILE', id, file.name);
    for (const key of keys) {
      try { await this.objects.delete(key); }
      catch (e) { console.error(`Object cleanup required ${key}`, e.message); }
    }
  }

  /** Портовый эквивалент migrateExisting(): дошифровка старых объектов и создание v1. */
  async migrateExisting() {
    for (const file of await this.files.findAll()) {
      let stored;
      try { stored = await this.objects.get(file.objectKey); } catch { continue; }
      const plain = this.crypto.decrypt(stored);
      const patch = {};
      if (!this.crypto.isEncrypted(stored)) await this.objects.put(file.objectKey, this.crypto.encrypt(plain));
      if (!file.checksum) patch.checksum = CryptoService.checksum(plain);
      if (!file.mimeType) patch.mimeType = StorageService.mime(file.name);
      if (!file.createdById) patch.createdById = file.ownerId;
      if (file.versionCount < 1) patch.versionCount = 1;
      if (Object.keys(patch).length) await this.files.update(file.id, patch);
      await this.ensureVersionRow(Object.assign(file, patch));
    }
  }
}

/* TOTP по RFC 6238 (порт TotpService.java): генерация секрета, проверка кода ±1 окно, otpauth URI. */
'use strict';

const crypto = require('node:crypto');

const BASE32_ALPHABET = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ234567';

class TotpService {
  newSecret() { return TotpService.encodeBase32(crypto.randomBytes(20)); }

  valid(secret, supplied) {
    if (!supplied || !/^\d{6}$/.test(String(supplied))) return false;
    const window = Math.floor(Date.now() / 30000);
    for (const offset of [-1, 0, 1]) {
      if (TotpService.safeEquals(this.code(secret, window + offset), String(supplied))) return true;
    }
    return false;
  }

  uri(email, secret) {
    const label = encodeURIComponent(`SkyShelf:${email}`);
    return `otpauth://totp/${label}?secret=${secret}&issuer=SkyShelf&algorithm=SHA1&digits=6&period=30`;
  }

  code(secret, counter) {
    const key = TotpService.decodeBase32(secret);
    const data = Buffer.alloc(8);
    data.writeBigUInt64BE(BigInt(counter));
    const hash = crypto.createHmac('sha1', key).update(data).digest();
    const offset = hash[hash.length - 1] & 0x0f;
    const binary = (hash.readUInt32BE(offset) & 0x7fffffff) % 1_000_000;
    return String(binary).padStart(6, '0');
  }

  static encodeBase32(bytes) {
    let out = '', buffer = 0, bits = 0;
    for (const value of bytes) {
      buffer = (buffer << 8) | value; bits += 8;
      while (bits >= 5) { out += BASE32_ALPHABET[(buffer >>> (bits - 5)) & 31]; bits -= 5; }
    }
    if (bits > 0) out += BASE32_ALPHABET[(buffer << (5 - bits)) & 31];
    return out;
  }

  static decodeBase32(value) {
    const chunks = [];
    let bits = '';
    for (const ch of String(value).replace(/=/g, '').toUpperCase()) {
      const index = BASE32_ALPHABET.indexOf(ch);
      if (index < 0) throw new Error('base32');
      bits += index.toString(2).padStart(5, '0');
    }
    for (let i = 0; i + 8 <= bits.length; i += 8) chunks.push(parseInt(bits.slice(i, i + 8), 2));
    return Buffer.from(chunks);
  }

  static safeEquals(a, b) {
    const ba = Buffer.from(a), bb = Buffer.from(b);
    return ba.length === bb.length && crypto.timingSafeEqual(ba, bb);
  }
}

/* Серверные сессии HttpOnly + CSRF-токены (порт HttpSessionSecurityContextRepository + CsrfToken).
   Хранилище в памяти достаточно для одного стенда; интерфейс SessionStore позволяет заменить его на БД. */
'use strict';

const crypto = require('node:crypto');

class Session {
  constructor(id) {
    this.id = id;
    this.userId = null;
    this.csrfToken = crypto.randomBytes(32).toString('hex');
    this.createdAt = Date.now();
    this.lastTouchedAt = Date.now();
  }
}

class SessionStore {
  constructor({ ttlMs = 30 * 60 * 1000 } = {}) {
    this.ttlMs = ttlMs;
    this.sessions = new Map();
    this.timer = setInterval(() => this.sweep(), 60_000);
    this.timer.unref?.();
  }

  sweep() {
    const now = Date.now();
    for (const [id, session] of this.sessions) {
      if (now - session.lastTouchedAt > this.ttlMs) this.sessions.delete(id);
    }
  }

  create() {
    const session = new Session(crypto.randomUUID());
    this.sessions.set(session.id, session);
    return session;
  }

  get(id) {
    const session = id ? this.sessions.get(id) : undefined;
    if (!session) return null;
    if (Date.now() - session.lastTouchedAt > this.ttlMs) { this.sessions.delete(session.id); return null; }
    session.lastTouchedAt = Date.now();
    return session;
  }

  destroy(id) { this.sessions.delete(id); }

  /** Смена session ID при входе (защита от фиксации сессии). */
  rotate(oldId) {
    if (oldId) this.sessions.delete(oldId);
    return this.create();
  }

  /** Все сессии пользователя (для инвалидации при блокировке/выходе). */
  byUser(userId) {
    return [...this.sessions.values()].filter((s) => s.userId === userId);
  }
}

/* Хранилище бинарных объектов (порт ObjectStore): два драйвера за общим интерфейсом —
   LocalObjectStore (файловая система, атомарная запись) и MinioObjectStore (S3 API через fetch).
   Минимизация зависимостей вместо тяжёлого SDK: подпись AWS SigV4 реализована локально. */
'use strict';

const fs = require('node:fs/promises');
const path = require('node:path');
const crypto = require('node:crypto');
const { NotFoundError } = require('../domain/errors');

/** Интерфейс объектного хранилища. */
class ObjectStore {
  // eslint-disable-next-line no-unused-vars
  async put(key, buffer) { throw new Error('not implemented'); }
  async get(key) { throw new Error('not implemented'); }
  // eslint-disable-next-line no-unused-vars
  async delete(key) { throw new Error('not implemented'); }
  static validateKey(key) {
    if (!/^[a-f0-9-]{36}$/.test(String(key))) throw new NotFoundError('Объект не найден или недоступен.');
    return key;
  }
}

class LocalObjectStore extends ObjectStore {
  constructor(dir) {
    super();
    this.root = path.resolve(dir);
  }

  async init() { await fs.mkdir(this.root, { recursive: true }); }

  path(key) {
    return path.join(this.root, ObjectStore.validateKey(key));
  }

  async put(key, buffer) {
    const target = this.path(key);
    const tmp = path.join(this.root, `upload-${crypto.randomUUID()}.tmp`);
    await fs.writeFile(tmp, buffer);
    try { await fs.rename(tmp, target); } finally { await fs.rm(tmp, { force: true }); }
  }

  async get(key) {
    try { return await fs.readFile(this.path(key)); }
    catch (e) { if (e.code === 'ENOENT') throw new NotFoundError('Объект не найден или недоступен.'); throw e; }
  }

  async delete(key) { await fs.rm(this.path(key), { force: true }); }
}

/** Минимальный клиент S3 (SigV4) для MinIO: PUT/GET/HEAD/DELETE объекта и создание бакета. */
class MinioObjectStore extends ObjectStore {
  constructor({ endpoint, accessKey, secretKey, bucket }) {
    super();
    this.base = new URL(endpoint);
    this.accessKey = accessKey;
    this.secretKey = secretKey;
    this.bucket = bucket;
  }

  static hmac(key, data) { return crypto.createHmac('sha256', key).update(data).digest(); }
  static sha256hex(data) { return crypto.createHash('sha256').update(data).digest('hex'); }

  async init() {
    // HEAD bucket → при 404 создаём
    const head = await this.request('HEAD', `/${this.bucket}/`, null, true);
    if (head.status === 404) await this.request('PUT', `/${this.bucket}/`, null, true);
  }

  sign(method, objectPath, body, extraHeaders = {}) {
    const host = this.base.host;
    const now = new Date();
    const amzDate = now.toISOString().replace(/[:-]|\.\d{3}/g, '');
    const dateStamp = amzDate.slice(0, 8);
    const payloadHash = body == null ? MinioObjectStore.sha256hex('') : MinioObjectStore.sha256hex(body);
    const headers = { host, 'x-amz-date': amzDate, 'x-amz-content-sha256': payloadHash, ...extraHeaders };
    const sortedNames = Object.keys(headers).sort();
    const canonicalHeaders = sortedNames.map((h) => `${h}:${String(headers[h]).trim()}\n`).join('');
    const signedHeaders = sortedNames.join(';');
    const canonicalRequest = [method, objectPath.split('?')[0].split('/').map(encodeURIComponent).join('/'), '', canonicalHeaders, signedHeaders, payloadHash].join('\n');
    const scope = `${dateStamp}/s3/aws4_request`;
    const stringToSign = ['AWS4-HMAC-SHA256', amzDate, scope, MinioObjectStore.sha256hex(canonicalRequest)].join('\n');
    const kDate = MinioObjectStore.hmac(`AWS4${this.secretKey}`, dateStamp);
    const kRegion = MinioObjectStore.hmac(kDate, 'us-east-1');
    const kService = MinioObjectStore.hmac(kRegion, 's3');
    const kSigning = MinioObjectStore.hmac(kService, 'aws4_request');
    const signature = crypto.createHmac('sha256', kSigning).update(stringToSign).digest('hex');
    headers.Authorization = `AWS4-HMAC-SHA256 Credential=${this.accessKey}/${scope}, SignedHeaders=${signedHeaders}, Signature=${signature}`;
    return headers;
  }

  async request(method, objectPath, body, silentStatuses = false) {
    const url = new URL(objectPath, this.base);
    const res = await fetch(url, { method, body: body ?? undefined, headers: this.sign(method, objectPath, body) });
    if (!res.ok && !silentStatuses && res.status !== 404) throw new Error(`MinIO ${method} ${objectPath}: HTTP ${res.status}`);
    return res;
  }

  async put(key, buffer) {
    ObjectStore.validateKey(key);
    const res = await this.request('PUT', `/${this.bucket}/${key}`, buffer);
    if (!res.ok) throw new Error(`MinIO put failed: HTTP ${res.status}`);
  }

  async get(key) {
    ObjectStore.validateKey(key);
    const res = await this.request('GET', `/${this.bucket}/${key}`, null, true);
    if (res.status === 404) throw new NotFoundError('Объект не найден или недоступен.');
    if (!res.ok) throw new Error(`MinIO get failed: HTTP ${res.status}`);
    return Buffer.from(await res.arrayBuffer());
  }

  async delete(key) {
    ObjectStore.validateKey(key);
    await this.request('DELETE', `/${this.bucket}/${key}`, null, true);
  }
}

/** Фабрика драйверов (OCP: новый драйвер = новый класс без правки остальных). */
function createObjectStore(config) {
  if (config.storageDriver === 'minio') {
    return new MinioObjectStore({
      endpoint: config.minio.endpoint, accessKey: config.minio.accessKey,
      secretKey: config.minio.password, bucket: config.minio.bucket,
    });
  }
  return new LocalObjectStore(config.storageDir);
}

/* Простой версионируемый миграционный движок (аналог Flyway): файлы src/db/migrations/*.sql
   применяются по порядку и записываются в таблицу schema_migrations. */
'use strict';

const fs = require('node:fs');
const path = require('node:path');

class MigrationRunner {
  constructor(pool, dir = path.join(__dirname, 'migrations')) {
    this.pool = pool;
    this.dir = dir;
  }

  files() {
    return fs.readdirSync(this.dir).filter((f) => f.endsWith('.sql')).sort();
  }

  async applied() {
    await this.pool.query(`CREATE TABLE IF NOT EXISTS schema_migrations (
      name text PRIMARY KEY, applied_at timestamptz NOT NULL DEFAULT now())`);
    const { rows } = await this.pool.query('SELECT name FROM schema_migrations ORDER BY name');
    return new Set(rows.map((r) => r.name));
  }

  /** Применяет все ещё не выполненные миграции. Возвращает список применённых файлов. */
  async up() {
    const done = await this.applied();
    const pending = this.files().filter((f) => !done.has(f));
    for (const file of pending) {
      const sql = fs.readFileSync(path.join(this.dir, file), 'utf8');
      const connection = await this.pool.connect();
      try {
        await connection.query('BEGIN');
        await connection.query(sql);
        await connection.query('INSERT INTO schema_migrations (name) VALUES ($1)', [file]);
        await connection.query('COMMIT');
        console.log(`migrated: ${file}`);
      } catch (error) {
        await connection.query('ROLLBACK');
        throw new Error(`Migration ${file} failed: ${error.message}`);
      } finally {
        connection.release();
      }
    }
    return pending;
  }

  async status() {
    const done = await this.applied();
    return this.files().map((f) => ({ file: f, applied: done.has(f) }));
  }
}

module.exports = { MigrationRunner };

/* Пул соединений PostgreSQL и транзакционный контекст (AsyncLocalStorage).
   Репозитории берут клиент через currentClient(): внутри транзакции — это connection,
   вне транзакции — пул. Порт JPA EntityManager + @Transactional. */
'use strict';

const { AsyncLocalStorage } = require('node:async_hooks');
const { Pool } = require('pg');

class Database {
  constructor(connectionString, { max = 10, ssl = false } = {}) {
    this.pool = new Pool({ connectionString, max, ssl: ssl ? { rejectUnauthorized: false } : false });
    this.storage = new AsyncLocalStorage();
  }

  /** Текущий клиент БД (транзакция или пул). */
  get client() { return this.storage.getStore() || this.pool; }

  query(text, params) { return this.client.query(text, params); }

  /** Запуск кода в транзакции SELECT ... FOR UPDATE / COMMIT / ROLLBACK. */
  async transaction(fn) {
    const existing = this.storage.getStore();
    if (existing) return fn(existing); // вложенная транзакция переиспользует connection
    const connection = await this.pool.connect();
    try {
      await connection.query('BEGIN');
      const result = await this.storage.run(connection, () => fn(connection));
      await connection.query('COMMIT');
      return result;
    } catch (error) {
      try { await connection.query('ROLLBACK'); } catch { /* соединение уже разорвано */ }
      throw error;
    } finally {
      connection.release();
    }
  }

  async close() { await this.pool.end(); }
}

'use strict';

const Fastify = require('fastify');
const fastifyCookie = require('@fastify/cookie');
const fastifyMultipart = require('@fastify/multipart');

const { AppConfig } = require('./config');
const { Database } = require('./db/database');
const { Repositories } = require('./repositories');
const { CryptoService } = require('./services/crypto-service');
const { SessionStore } = require('./services/session-store');
const { TotpService } = require('./services/totp-service');
const { StorageService } = require('./services/storage-service');
const { AuditService } = require('./services/audit-service');
const { AuthService } = require('./services/auth-service');
const { ProjectService } = require('./services/project-service');
const { createObjectStore, LocalObjectStore } = require('./services/object-store');
const { AuthController, SESSION_COOKIE } = require('./controllers/auth-controller');

async function buildApp(config = new AppConfig()) {
  const db = new Database(config.dbUrl);
  const repos = new Repositories(db);

  const crypto = new CryptoService(config.masterKey);
  const sessions = new SessionStore({ ttlMs: config.sessionTtlMs });
  const totp = new TotpService();
  const audit = new AuditService(repos);
  const auth = new AuthService({ accounts: repos.accounts, crypto, totp, audit, sessions });
  const projects = new ProjectService({
    projects: repos.projects, memberships: repos.memberships, accounts: repos.accounts,
    tasks: repos.tasks, files: repos.files, submissions: repos.submissions,
    crypto, audit,
  });
  const storage = new StorageService({
    files: repos.files, folders: repos.folders, shares: repos.shares, accounts: repos.accounts,
    versions: repos.versions, projectsRepo: repos.projects,
    objects: createObjectStoreSafe(config), crypto, audit, projectService: projects, config,
  });

  const services = { config, db, repos, crypto, sessions, totp, audit, auth, projects, storage };

  const app = Fastify({ logger: { level: process.env.LOG_LEVEL || 'info' } });
  await app.register(fastifyCookie);
  await app.register(fastifyMultipart, { limits: { fileSize: config.maxUploadBytes } });

  // восстановление сессии из HttpOnly-куки для каждого запроса
  app.addHook('onRequest', async (request) => {
    const id = request.cookies[SESSION_COOKIE];
    request.session = id ? sessions.get(id) : null;
    if (request.session) request.session.lastTouchedAt = Date.now();
  });

  // CSRF-проверка для изменяющих запросов (аналог Spring Security CsrfToken)
  app.addHook('preHandler', async (request, reply) => {
    if (!['POST', 'PUT', 'PATCH', 'DELETE'].includes(request.method)) return;
    // регистрация/вход выполняются до установления сессии — CSRF здесь не применим
    if (request.url === '/api/auth/register' || request.url === '/api/auth/login') return;
    const token = request.headers['x-csrf-token'];
    if (!request.session || !token || token !== request.session.csrfToken) {
      return reply.code(403).type('application/json').send({ message: 'CSRF-токен недействителен.' });
    }
  });

  // /api/health объявляет AuthController — дубль роута роняет Fastify (FST_ERR_DUPLICATED_ROUTE)
  new AuthController(services).register(app);

  app.setErrorHandler((error, request, reply) => {
    if (error.statusCode >= 500) request.log.error(error);
    return reply.code(error.statusCode || 500).type('application/json')
      .send({ message: error.statusCode ? error.message : 'Внутренняя ошибка сервера.' });
  });

  return { app, db, services };
}

/* Auth + профиль (порт AuthController.java): регистрация, вход с TOTP, HttpOnly-сессии, CSRF, /me. */
'use strict';

const { Controller } = require('./controller');
const { UnauthorizedError, BadRequestError } = require('../domain/errors');

const SESSION_COOKIE = 'SKYSHELF_SESSION';

class AuthController extends Controller {
  routes() {
    return [
      ['GET', '/api/health', this.health],
      ['GET', '/api/auth/csrf', this.csrf],
      ['POST', '/api/auth/register', this.signup],
      ['POST', '/api/auth/login', this.login],
      ['POST', '/api/auth/logout', this.logout],
      ['GET', '/api/me', this.me],
      ['PATCH', '/api/me', this.patchMe],
    ];
  }

  static sessionCookie(value, maxAge = 0) {
    return `${SESSION_COOKIE}=${value}; Path=/; HttpOnly; SameSite=Lax; Max-Age=${maxAge}`;
  }

  health() {
    return { status: 'ok', version: '1.0.0', encryption: 'AES-256-GCM' };
  }

  csrf(request, reply) {
    const session = request.session ?? this.services.sessions.create();
    reply.header('Set-Cookie', AuthController.sessionCookie(session.id, this.services.config.sessionTtlMs / 1000));
    return { token: session.csrfToken, header: 'X-CSRF-Token' };
  }

  // не «register»: это имя занято Controller.register(app), который вешает роуты на Fastify
  async signup(request, reply) {
    const account = await this.services.auth.register(request.body ?? {}, request.ip);
    reply.code(201);
    return account.view();
  }

  async login(request, reply) {
    const body = request.body ?? {};
    if (typeof body.email !== 'string' || typeof body.password !== 'string') throw new BadRequestError('Проверьте поля формы.');
    const { auth, sessions } = this.services;
    const account = await auth.authenticate(body.email, body.password, request.ip);
    if (account.twoFactorEnabled) await auth.verifyTwoFactor(account, body.code);
    // защита от фиксации сессии: старый session ID уничтожается, создаётся новый
    const session = sessions.rotate(request.session?.id ?? request.cookies[SESSION_COOKIE_ID]);
    session.userId = account.id;
    reply.header('Set-Cookie', AuthController.sessionCookie(session.id, this.services.config.sessionTtlMs / 1000));
    await auth.touchLogin(account);
    return account.view();
  }

  logout(request, reply) {
    if (request.session) this.services.sessions.destroy(request.session.id);
    reply.header('Set-Cookie', AuthController.sessionCookie('', 0));
    return { ok: true };
  }

  async me(request) {
    const account = await this.services.auth.current(request.session);
    return account.view();
  }

  async patchMe(request) {
    const account = await this.services.auth.current(request.session);
    const { name, theme, accent } = request.body ?? {};
    if (!name || !String(name).trim()) throw new BadRequestError('Имя обязательно.');
    if (!['dark', 'light'].includes(theme)) throw new BadRequestError('Выберите тему оформления.');
    if (!['blue', 'violet', 'mint'].includes(accent)) throw new BadRequestError('Выберите акцентный цвет.');
    await this.services.repos.accounts.update(account.id, {
      name: String(name).trim().slice(0, 80), theme, accent,
    });
    await this.services.audit.record(account.id, null, 'PROFILE_UPDATED', 'ACCOUNT', account.id, 'Профиль и оформление');
    const updated = await this.services.repos.accounts.findById(account.id);
    return updated.view();
  }
}

/* Контроллер — базовый класс HTTP-слоя (аналог @RestController): объявляет route-таблицу,
   регистрирует её в Fastify и оборачивает обработчик единой обработкой ошибок. */
'use strict';

const { DomainError } = require('../domain/errors');

class Controller {
  constructor(services) { this.services = services; }

  /** [ [method, path, handler], ... ] — переопределяется потомками. */
  routes() { return []; }

  register(app) {
    for (const [method, path, handler] of this.routes()) {
      app.route({
        method, url: path,
        handler: async (request, reply) => {
          try {
            const result = await handler.call(this, request, reply);
            if (!reply.sent) reply.send(result ?? null);
          } catch (error) {
            this.handleError(request, reply, error);
          }
        },
      });
    }
  }

  handleError(request, reply, error) {
    if (error instanceof DomainError) {
      return reply.code(error.status).type('application/json').send({ message: error.message });
    }
    if (error && error.rawBody === undefined && error.code === 'FST_REQ_FILE_TOO_LARGE') {
      return reply.code(413).type('application/json').send({ message: 'Максимальный размер файла — 20 МБ.' });
    }
    if (error && (error.code === 'FST_VALIDATION_ERROR' || error.statusCode === 400 && /body|JSON/i.test(String(error.message)))) {
      return reply.code(400).type('application/json').send({ message: 'Некорректный запрос.' });
    }
    if (error && error.code === '23505') {
      return reply.code(409).type('application/json').send({ message: 'Такая запись уже существует.' });
    }
    request.log.error({ err: error }, 'Request failed');
    return reply.code(500).type('application/json').send({ message: 'Не удалось выполнить действие. Проверьте журнал сервера.' });
  }

  static badRequest(message) { return new DomainError(400, message); }
}

/* Абстрактный репозиторий (DIP): сервисы зависят от этого интерфейса, а не от SQL.
   Потомки объявляют таблицу, сущность и поле поиска; CRUD — общий. */
'use strict';

const crypto = require('node:crypto');
const { Entity } = require('../domain/entities');

class Repository {
  constructor(db, { table, Entity, searchColumn = null }) {
    this.db = db;
    this.table = table;
    this.Entity = Entity;
    this.searchColumn = searchColumn;
  }

  get client() { return this.db.client; }

  static columnsOf(Entity) { return Object.values(Entity.fieldMap); }

  /** Разрешаемые set-колонки для update(): только существующие колонки таблицы. */
  allowedColumns() { return new Set(Repository.columnsOf(this.Entity)); }

  async findById(id) {
    const { rows } = await this.client.query(`SELECT * FROM ${this.table} WHERE id = $1`, [id]);
    return rows[0] ? new this.Entity(rows[0]) : null;
  }

  async findAll() {
    const { rows } = await this.client.query(`SELECT * FROM ${this.table}`);
    return rows.map((r) => new this.Entity(r));
  }

  async count(where = null, params = []) {
    const sql = `SELECT count(*)::bigint AS c FROM ${this.table}${where ? ` WHERE ${where}` : ''}`;
    const { rows } = await this.client.query(sql, params);
    return rows[0].c;
  }

  async exists(where, params = []) {
    const { rows } = await this.client.query(`SELECT 1 FROM ${this.table} WHERE ${where} LIMIT 1`, params);
    return rows.length > 0;
  }

  async insert(entity, { returningId = true } = {}) {
    if (!(entity instanceof Entity)) entity = new this.Entity(entity); // допускаем plain-объекты
    const row = entity.toRow();
    if (!row.id && this.table !== 'share_links' && this.table !== 'submission_links') row.id = crypto.randomUUID();
    const cols = Object.keys(row);
    const values = Object.values(row);
    const placeholders = cols.map((_, i) => `$${i + 1}`).join(', ');
    const sql = `INSERT INTO ${this.table} (${cols.join(', ')}) VALUES (${placeholders})${returningId && row.id ? ' RETURNING id' : ''}`;
    const res = await this.client.query(sql, values);
    entity.id = row.id ?? res.rows?.[0]?.id;
    return entity;
  }

  /** Частичное обновление: поля сущности → колонки, whitelist по схеме таблицы. */
  async update(id, patch, extraWhere = '', extraParams = []) {
    const map = this.Entity.fieldMap;
    const allowed = this.allowedColumns();
    const sets = [];
    const params = [];
    for (const [field, value] of Object.entries(patch)) {
      const column = map[field] ?? field;
      if (!allowed.has(column)) throw new Error(`Unknown column "${column}" for ${this.table}`);
      params.push(value);
      sets.push(`${column} = $${params.length}`);
    }
    if (!sets.length) return this.findById(id);
    params.push(id, ...extraParams);
    const sql = `UPDATE ${this.table} SET ${sets.join(', ')} WHERE id = $${params.length - extraParams.length}${extraWhere ? ` AND ${extraWhere}` : ''}`;
    const res = await this.client.query(sql, params);
    return res.rowCount > 0;
  }

  async deleteById(id, extraWhere = '', extraParams = []) {
    const params = [id, ...extraParams];
    const sql = `DELETE FROM ${this.table} WHERE id = $1${extraWhere ? ` AND ${extraWhere}` : ''}`;
    const res = await this.client.query(sql, params);
    return res.rowCount > 0;
  }

  async deleteWhere(where, params = []) {
    const res = await this.client.query(`DELETE FROM ${this.table} WHERE ${where}`, params);
    return res.rowCount;
  }

  async findWhere(where, params = [], orderBy = '') {
    const { rows } = await this.client.query(`SELECT * FROM ${this.table} WHERE ${where}${orderBy ? ` ORDER BY ${orderBy}` : ''}`, params);
    return rows.map((r) => new this.Entity(r));
  }

  /** Блокировка строки на время транзакции (порт @Lock(PESSIMISTIC_WRITE)). */
  async lockById(id) {
    const { rows } = await this.client.query(`SELECT * FROM ${this.table} WHERE id = $1 FOR UPDATE`, [id]);
    return rows[0] ? new this.Entity(rows[0]) : null;
  }
}

   API возвращает отдельные DTO (view-методы), а не эти объекты — как в Java-версии. */
'use strict';

const { BadRequestError } = require('./errors');

/** Базовая сущность: маппинг строк БД (snake_case) в поля домена (camelCase). */
class Entity {
  static columns = [];
  constructor(row = {}) { Object.assign(this, this.constructor.fromRow(row)); }
  static fromRow(row) {
    const out = {};
    // this — класс-наследник: берём его собственный fieldMap (статика не наследуется как значение)
    // принимаем и строку БД (snake_case), и plain-объект домена (camelCase):
    // иначе поля вроде passwordHash/projectId молча теряются при insert()
    for (const [field, column] of Object.entries(this.fieldMap ?? Entity.fieldMap)) {
      out[field] = column in row ? row[column] : row[field];
    }
    return out;
  }
  toRow() {
    const out = {};
    for (const [field, column] of Object.entries(this.constructor.fieldMap)) if (this[field] !== undefined) out[column] = this[field];
    return out;
  }
}

/** Строгие правила имён: без управляемых символов, «/», «\\», «.» и «..». */
function validateName(value, max) {
  if (value == null) throw new BadRequestError('Укажите название.');
  const clean = String(value).trim();
  if (!clean || clean.length > max || clean === '.' || clean === '..' || /[\p{Cc}\/\\]/u.test(clean)) {
    throw new BadRequestError('Недопустимое название.');
  }
  return clean;
}

/* Доменные ошибки уровня сервиса (порт ResponseStatusException из Java-версии). */
'use strict';

class DomainError extends Error {
  constructor(status, message) {
    super(message);
    this.name = this.constructor.name;
    this.status = status;
  }
}

class BadRequestError extends DomainError { constructor(m = 'Недопустимый запрос.') { super(400, m); } }
class UnauthorizedError extends DomainError { constructor(m = 'Войдите в аккаунт') { super(401, m); } }
class ForbiddenError extends DomainError { constructor(m = 'Доступ запрещён или сессия устарела. Обновите страницу.') { super(403, m); } }
class NotFoundError extends DomainError { constructor(m = 'Объект не найден или недоступен.') { super(404, m); } }
class ConflictError extends DomainError { constructor(m = 'Такая запись уже существует.') { super(409, m); } }
class PayloadTooLargeError extends DomainError { constructor(m = 'Максимум 20 МБ на файл.') { super(413, m); } }
class TooManyRequestsError extends DomainError { constructor(m = 'Слишком много попыток. Попробуйте через 10 минут.') { super(429, m); } }
class IntegrityError extends DomainError { constructor(m = 'Проверка целостности файла не пройдена.') { super(500, m); } }

module.exports = {
  DomainError, BadRequestError, UnauthorizedError, ForbiddenError,
  NotFoundError, ConflictError, PayloadTooLargeError, TooManyRequestsError, IntegrityError,
};
